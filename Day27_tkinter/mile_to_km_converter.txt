import tkinter 

window = tkinter.Tk()
window.title("Mile to KM Converter")
window.minsize(width=500, height=300)
window.config(padx=20, pady=30)

def calculate():
    miles = float(miles_input.get()) # Get the value from the input field and convert it to number
    km = miles * 1.60934 # Convert miles to kilometers
    result_label.config(text=f"{km:.2f}") # Update the result label with the converted value formatted to 2 decimal places
    miles_input.delete(0, tkinter.END) # Clear the input field after calculation


mile_label = tkinter.Label(text="Miles")
mile_label.grid(row=0, column=2)

miles_input = tkinter.Entry()
miles_input.grid(row=0, column=1)

equal_label = tkinter.Label(text="is equal to")
equal_label.grid(row=1, column=0)

km_label = tkinter.Label(text="KM")
km_label.grid(row=1, column=2)

result_label = tkinter.Label(text="0")
result_label.grid(row=1, column=1)

calculate_button = tkinter.Button(text="Calculate", command=calculate)
calculate_button.grid(row=2, column=1)


window.mainloop()
